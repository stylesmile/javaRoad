# oracle 创建用户
```sqll
ALTER SESSION SET CONTAINER = ORCLPDB1;

DECLARE
  v_count NUMBER;
BEGIN
  SELECT COUNT(*) INTO v_count FROM dba_users WHERE username = 'JL_APP_PLATFORM';
  IF v_count = 0 THEN
    EXECUTE IMMEDIATE 'CREATE USER JL_APP_PLATFORM IDENTIFIED BY Juleitech12345';
  ELSE
    EXECUTE IMMEDIATE 'ALTER USER JL_APP_PLATFORM IDENTIFIED BY Juleitech12345';
  END IF;

  EXECUTE IMMEDIATE 'ALTER USER JL_APP_PLATFORM ACCOUNT UNLOCK';
  EXECUTE IMMEDIATE 'GRANT CREATE SESSION TO JL_APP_PLATFORM';
  EXECUTE IMMEDIATE 'GRANT CONNECT, RESOURCE TO JL_APP_PLATFORM';
  EXECUTE IMMEDIATE 'GRANT CREATE VIEW, CREATE SEQUENCE, CREATE TRIGGER TO JL_APP_PLATFORM';
  EXECUTE IMMEDIATE 'GRANT UNLIMITED TABLESPACE TO JL_APP_PLATFORM';
  EXECUTE IMMEDIATE 'ALTER USER JL_APP_PLATFORM QUOTA UNLIMITED ON USERS';
END;

```
